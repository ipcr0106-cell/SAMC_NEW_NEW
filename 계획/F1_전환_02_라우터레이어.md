### `_fetch_pipeline_step`

**현재:**
```python
async def _fetch_pipeline_step(
    db: asyncpg.Connection, case_id: str, step_key: str = "1"
) -> Optional[asyncpg.Record]:
    return await db.fetchrow("""
        SELECT id, case_id, step_key, step_name, status,
               ai_result, final_result, edit_reason,
               law_references, created_at, updated_at
          FROM pipeline_steps
         WHERE case_id = $1 AND step_key = $2
         LIMIT 1
    """, case_id, step_key)
```

**전환 후:**
```python
def _fetch_pipeline_step(
    case_id: str, step_key: str = "1"
) -> Optional[dict]:
    supabase = get_supabase()
    result = supabase.table("pipeline_steps") \
        .select(
            "id, case_id, step_key, step_name, status, "
            "ai_result, final_result, edit_reason, "
            "law_references, created_at, updated_at"
        ) \
        .eq("case_id", case_id) \
        .eq("step_key", step_key) \
        .limit(1) \
        .execute()
    return result.data[0] if result.data else None
```

### `_upsert_pipeline_step`

**현재:**
```python
async def _upsert_pipeline_step(
    db: asyncpg.Connection,
    case_id: str,
    status: str,
    ai_result: dict,
) -> None:
    await db.execute("""
        INSERT INTO pipeline_steps (case_id, step_key, step_name, status, ai_result)
        VALUES ($1, '1', 'import_check', $2, $3::jsonb)
        ON CONFLICT (case_id, step_key) DO UPDATE
          SET status = EXCLUDED.status,
              ai_result = EXCLUDED.ai_result,
              updated_at = NOW()
    """, case_id, status, json.dumps(ai_result, ensure_ascii=False))
```

**전환 후:**
```python
def _upsert_pipeline_step(
    case_id: str,
    status: str,
    ai_result: dict,
) -> None:
    supabase = get_supabase()
    supabase.table("pipeline_steps").upsert(
        {
            "case_id": case_id,
            "step_key": "1",
            "step_name": "import_check",
            "status": status,
            "ai_result": ai_result,  # supabase-py가 dict를 JSONB로 자동 직렬화
        },
        on_conflict="case_id,step_key"
    ).execute()
```

> `json.dumps` 불필요 — supabase-py는 dict를 JSONB로 자동 처리.

### `_record_to_json`

**현재:** asyncpg.Record의 JSONB 컬럼을 수동 파싱

```python
def _record_to_json(row: asyncpg.Record, field: str) -> Any:
    raw = row[field]
    if raw is None:
        return None
    if isinstance(raw, (dict, list)):
        return raw
    try:
        return json.loads(raw)
    except (TypeError, ValueError):
        return None
```

**전환 후:** supabase-py는 JSONB를 dict로 자동 반환하므로 단순화

```python
def _record_to_json(row: dict, field: str) -> Any:
    return row.get(field)  # 이미 dict 또는 None
```

```python
async def confirm_feature1(case_id: str) -> dict:
```

### `_check_ownership`

**현재:**
```python
async def _check_ownership(
    db: asyncpg.Connection, table: str, row_id: str, user_uuid: UUID
) -> None:
    row_uuid = _parse_uuid(row_id)
    row = await db.fetchrow(f"SELECT created_by FROM {table} WHERE id = $1", row_uuid)
    if not row:
        raise HTTPException(404, ...)
    if row["created_by"] is None or row["created_by"] != user_uuid:
        raise HTTPException(403, ...)
```

**전환 후:**
```python
def _check_ownership(table: str, row_id: str, user_uuid: UUID) -> None:
    row_uuid = _parse_uuid(row_id)
    supabase = get_supabase()
    result = supabase.table(table) \
        .select("created_by") \
        .eq("id", str(row_uuid)) \
        .limit(1).execute()
    if not result.data:
        raise HTTPException(404, detail={"error": "NOT_FOUND", ...})
    created_by = result.data[0].get("created_by")
    if created_by is None or created_by != str(user_uuid):
        raise HTTPException(403, detail={"error": "FORBIDDEN", ...})
```

> UUID 비교: supabase-py는 UUID를 str로 반환 → `str(user_uuid)`로 비교.

### `GET /{table}` — `list_rows`

**현재:** 동적 SQL + `LIMIT/OFFSET`

```python
async def list_rows(..., db: asyncpg.Connection = Depends(get_conn_dep)) -> dict:
    rows = await db.fetch(query, *params)
    total = await db.fetchval(f"SELECT COUNT(*) FROM {table} {where}", *filter_params)
```

**전환 후:**

```python
async def list_rows(
    table: TableName,
    only_mine: bool = Query(False),
    only_unverified: bool = Query(False),
    limit: int = Query(200, ge=1, le=1000),
    offset: int = Query(0, ge=0),
    x_user_id: Optional[str] = Header(None, alias="X-User-Id"),
) -> dict:
    meta = _require_table(table)
    user_uuid = _user_id(x_user_id)
    supabase = get_supabase()

    # 카운트 포함 쿼리 (count="exact")
    query = supabase.table(table).select(meta["list_columns"], count="exact")

    if only_mine:
        if not user_uuid:
            raise HTTPException(400, detail={"error": "USER_ID_REQUIRED", ...})
        query = query.eq("created_by", str(user_uuid))
    if only_unverified:
        query = query.eq("is_verified", False)

    # 정렬 + 페이지네이션
    # order_by 예: "name_ko" 또는 "food_type, additive_name"
    order_cols = [c.strip() for c in meta["order_by"].split(",")]
    for col in order_cols:
        query = query.order(col)

    result = query.range(offset, offset + limit - 1).execute()

    return {
        "table": table,
        "total": result.count,
        "limit": limit,
        "offset": offset,
        "items": result.data,
    }
```

> `select(..., count="exact")` → `result.count`에 전체 건수 반환.

```python
async def create_row(...) -> dict:
    meta = _require_table(table)
    user_uuid = _user_id(x_user_id)
    if not user_uuid:
        raise HTTPException(401, ...)

    data = _filter_columns(body.data, meta["columns"])
    if not data:
        raise HTTPException(400, ...)

    result = supabase.table(table).insert(data).execute()
    new_id = result.data[0]["id"]
```
